132// ---------------------------------------------------------------------------
133// Relay routing (neurorelay.sillyprootsoda.com)
134// ---------------------------------------------------------------------------
135// The relay is a dumb pass-through WebSocket router keyed by room code - it never
136// looks at game message contents, only at these routing bytes/JSON control frames.
137//
138// Binary game messages (everything above this comment) get ONE extra leading byte
139// added/stripped by the relay itself, never by the game code on both ends symmetrically:
140// guest -> relay -> host : relay PREPENDS the sender's slot (1..MAX_NET_PLAYERS-1)
141// before forwarding to the host. Guests send raw, unprefixed.
142// host -> relay -> guest : host PREPENDS a target-slot byte (0 = broadcast to every
143// guest in the room, N = unicast to just that guest's slot).
144// The relay strips this byte before delivering to guest(s),
145// so guests always receive a clean, unprefixed payload.
146constexpr uint8_t RELAY_BROADCAST = 0;
147
148inline std::string ToRelayUnicast(uint8_t targetSlot, const std::string& raw) {
149 std::string out;
150 out.reserve(raw.size() + 1);
151 out.push_back(static_cast<char>(targetSlot));
152 out += raw;
153 return out;
154}
155
156// Host-side receive: split the relay's sender-slot prefix off a guest's message.
157inline uint8_t PeekSenderSlot(const std::string& data) {
158 return data.empty() ? 0 : static_cast<uint8_t>(data[0]);
159}
160
161inline std::string StripSenderSlot(const std::string& data) {
162 return data.size() > 1 ? data.substr(1) : std::string();
163}
